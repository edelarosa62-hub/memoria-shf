        // ===== Estado inicial =====
        let filaCounter = 1;

        function crearFilaIndiviso(clave, ubicacion, uso, privativa, accesoria, totalPriv, coef, comun, obs) {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td><input type="text" value="${clave || 'U-' + String(filaCounter).padStart(2, '0')}" placeholder="Clave"></td>
                <td><input type="text" value="${ubicacion || ''}" placeholder="Nivel / ubicación"></td>
                <td>
                    <select style="width:100%; padding:4px 6px; border:1px solid transparent; background:transparent; border-radius:4px; font-size:13px;">
                        <option value="Habitacional" ${uso === 'Habitacional' ? 'selected' : ''}>Habitacional</option>
                        <option value="Comercial" ${uso === 'Comercial' ? 'selected' : ''}>Comercial</option>
                        <option value="Estacionamiento" ${uso === 'Estacionamiento' ? 'selected' : ''}>Estacionamiento</option>
                        <option value="Bodega" ${uso === 'Bodega' ? 'selected' : ''}>Bodega</option>
                        <option value="Otro" ${uso === 'Otro' ? 'selected' : ''}>Otro</option>
                    </select>
                </td>
                <td><input type="number" step="0.01" class="campo-privativa" value="${privativa || '0.00'}" placeholder="0.00" onchange="calcularTotales()" oninput="calcularTotales()"></td>
                <td><input type="number" step="0.01" class="campo-accesoria" value="${accesoria || '0.00'}" placeholder="0.00" onchange="calcularTotales()" oninput="calcularTotales()"></td>
                <td><input type="number" step="0.01" class="campo-total-priv" value="${totalPriv || '0.00'}" placeholder="0.00" readonly style="background:#f1f4f8; text-align:center;"></td>
                <td><input type="number" step="0.01" class="campo-coef" value="${coef || '0.00'}" placeholder="0.00" readonly style="background:#f1f4f8; text-align:center;"></td>
                <td><input type="number" step="0.01" class="campo-comun" value="${comun || '0.00'}" placeholder="0.00" readonly style="background:#f1f4f8; text-align:center;"></td>
                <td><input type="text" value="${obs || ''}" placeholder="Observaciones"></td>
            `;
            filaCounter++;
            return tr;
        }

        function agregarFilaIndiviso(clave, ubicacion, uso, privativa, accesoria, totalPriv, coef, comun, obs) {
            const tbody = document.getElementById('tbodyIndivisos');
            const tr = crearFilaIndiviso(clave, ubicacion, uso, privativa, accesoria, totalPriv, coef, comun, obs);
            tbody.appendChild(tr);
            calcularTotales();
        }

        function eliminarUltimaFila() {
            const tbody = document.getElementById('tbodyIndivisos');
            if (tbody.children.length > 0) {
                tbody.removeChild(tbody.lastElementChild);
                calcularTotales();
            } else {
                alert('No hay filas para eliminar.');
            }
        }

        function calcularTotales() {
            const tbody = document.getElementById('tbodyIndivisos');
            const filas = tbody.querySelectorAll('tr');

            let sumPrivativa = 0;
            let sumAccesoria = 0;
            let sumTotalPriv = 0;
            let sumComun = 0;

            filas.forEach(tr => {
                const privInput = tr.querySelector('.campo-privativa');
                const accInput = tr.querySelector('.campo-accesoria');
                const totalPrivInput = tr.querySelector('.campo-total-priv');
                const coefInput = tr.querySelector('.campo-coef');
                const comunInput = tr.querySelector('.campo-comun');

                if (privInput && accInput && totalPrivInput && coefInput && comunInput) {
                    const priv = parseFloat(privInput.value) || 0;
                    const acc = parseFloat(accInput.value) || 0;
                    const total = priv + acc;
                    totalPrivInput.value = total.toFixed(2);

                    sumPrivativa += priv;
                    sumAccesoria += acc;
                    sumTotalPriv += total;
                }
            });

            // Recalcular coeficientes y área común (asumiendo que el total de áreas comunes se asigna proporcionalmente)
            // Para este ejemplo, mostramos solo los totales. En una implementación completa se calcularía cada coeficiente.
            filas.forEach(tr => {
                const totalPrivInput = tr.querySelector('.campo-total-priv');
                const coefInput = tr.querySelector('.campo-coef');
                const comunInput = tr.querySelector('.campo-comun');
                if (totalPrivInput && coefInput && comunInput && sumTotalPriv > 0) {
                    const total = parseFloat(totalPrivInput.value) || 0;
                    const coef = (total / sumTotalPriv) * 100;
                    coefInput.value = coef.toFixed(2);
                    // Área común asignada (ejemplo: se asigna un total de 100 m² de áreas comunes)
                    // En un caso real, el total de áreas comunes se tomaría de la Sección VI
                    const totalComunes = 100; // Valor de ejemplo
                    const comun = (total / sumTotalPriv) * totalComunes;
                    comunInput.value = comun.toFixed(2);
                }
            });

            // Actualizar totales en el footer
            document.getElementById('totalPrivativa').value = sumPrivativa.toFixed(2);
            document.getElementById('totalAccesoria').value = sumAccesoria.toFixed(2);
            document.getElementById('totalPrivativaSum').value = sumTotalPriv.toFixed(2);

            // Sumar coeficientes
            let sumCoef = 0;
            filas.forEach(tr => {
                const coefInput = tr.querySelector('.campo-coef');
                if (coefInput) {
                    sumCoef += parseFloat(coefInput.value) || 0;
                }
            });
            document.getElementById('totalCoef').value = sumCoef.toFixed(2);

            // Sumar áreas comunes asignadas
            let sumComunTotal = 0;
            filas.forEach(tr => {
                const comunInput = tr.querySelector('.campo-comun');
                if (comunInput) {
                    sumComunTotal += parseFloat(comunInput.value) || 0;
                }
            });
            document.getElementById('totalComun').value = sumComunTotal.toFixed(2);
        }

            calcularTotales();

  // 1. Cargar datos de la hoja cuando se abre la página
  window.addEventListener('DOMContentLoaded', () => {
    fetch(URL_APPS_SCRIPT)
      .then(response => response.json())
      .then(datos => {
        console.log("Datos cargados desde la hoja de cálculo:", datos);
        // Aquí puedes mapear 'datos' a la Tabla de Indivisos o campos del HTML
      })
      .catch(err => console.error("Error al cargar datos:", err));
  });

  // 2. Función para enviar un nuevo registro desde el HTML
  function enviarNuevoRegistro() {
    // Organiza los valores en el mismo orden de columnas que tu Google Sheets
    const nuevoRegistro = [
      new Date().toLocaleString(), // Fecha/Hora
      document.getElementById('claveAvaluo')?.value || '',
      document.getElementById('cuv')?.value || '',
      document.getElementById('proyecto')?.value || ''
    ];

    fetch(URL_APPS_SCRIPT, {
      method: "POST",
      mode: "no-cors",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ registro: nuevoRegistro })
    })
    .then(() => alert("Registro agregado correctamente a la hoja de cálculo."))
    .catch(err => alert("Error al guardar: " + err));
  }
</script>
</body>
</html>
